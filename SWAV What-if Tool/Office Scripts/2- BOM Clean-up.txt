/**
 * BOM EXPLODER — Office Script
 * ============================================================
 * Reads the "BOM Report Output" sheet from THIS workbook,
 * recursively explodes all parent→child paths across all levels,
 * locations, and BOM types, then writes the flat result to a
 * "FlatBOM" sheet as a named Table.
 *
 * HOW TO USE:
 *   1. Open BOM_Report_Output_-_SWAV_What_If_Tool.xlsx in Excel Online
 *   2. Go to Automate → New Script → paste this code → Save as "BOM Exploder"
 *   3. Click Run
 *   4. FlatBOM sheet will be created/refreshed automatically
 *
 * Source sheet columns (0-indexed):
 *   0  ParentSku
 *   1  ParentLocationName
 *   4  ChildSku
 *   5  ChildLocationName
 *   8  Qty
 *   9  EffStartDate
 *   10 EffEndDate
 *   16 BOMText01  (Primary | ALT)
 * ============================================================
 */
function main(workbook: ExcelScript.Workbook) {

  // ── Types ────────────────────────────────────────────────
  interface Edge {
    childSku: string;
    childLoc: string;
    qty: number;
    effStart: string;
    effEnd: string;
  }

  interface StackFrame {
    currentSku: string;
    currentLoc: string;
    cumulativeQty: number;
    level: number;
    ancestorPath: string;
  }

  interface FlatRow {
    TopLevelSku: string;
    ChildSku: string;
    ImmediateParentSku: string;
    CumulativeQty: number;
    Level: number;
    BOMType: string;
    LocationName: string;
    EffStartDate: string;
    EffEndDate: string;
  }

  // ── 1. Read source sheet ─────────────────────────────────
  // Try to find by the well-known prefix first. If the workbook arrives
  // with only ONE sheet (whose name may vary), use that sheet as a fallback.
  let sourceSheet = findSheetByPrefix(workbook, "BOM Report Output");
  if (!sourceSheet) {
    const allSheets = workbook.getWorksheets();
    if (allSheets.length === 1) {
      sourceSheet = allSheets[0];
      console.log(`No sheet starting with 'BOM Report Output' found. Falling back to the only sheet: "${sourceSheet.getName()}"`);
    } else {
      console.log(`ERROR: No sheet starting with 'BOM Report Output' found, and ${allSheets.length} sheets exist — cannot auto-select.`);
      return;
    }
  }
  console.log(`Using source sheet: "${sourceSheet.getName()}"`);

  const usedRange = sourceSheet.getUsedRange();
  if (!usedRange) {
    console.log("ERROR: Source sheet is empty.");
    return;
  }

  const totalRows = usedRange.getRowCount();
  console.log(`Source rows (incl. header): ${totalRows}`);

  // Column indices (0-based)
  const COL_PARENT_SKU = 0;
  const COL_PARENT_LOC = 1;
  const COL_CHILD_SKU  = 4;
  const COL_CHILD_LOC  = 5;
  const COL_QTY        = 8;
  const COL_EFF_START  = 9;
  const COL_EFF_END    = 10;
  const COL_BOM_TYPE   = 16;

  // Read in chunks of 2000 rows to avoid the internal getValues() memory limit
  const CHUNK = 2000;

  // ── 2. Build adjacency map ───────────────────────────────
  const adjMap = new Map<string, Edge[]>();
  const allParentSkus = new Set<string>();
  const allChildSkus  = new Set<string>();

  for (let startRow = 1; startRow < totalRows; startRow += CHUNK) {
    const rowCount = Math.min(CHUNK, totalRows - startRow);

    // Read only the 8 columns we need — avoids pulling all 27 columns
    const getCol = (col: number): (string | number | boolean)[] =>
      sourceSheet
        .getRangeByIndexes(startRow, col, rowCount, 1)
        .getValues()
        .map((r: (string | number | boolean)[]) => r[0]);

    const parentSkus = getCol(COL_PARENT_SKU);
    const parentLocs = getCol(COL_PARENT_LOC);
    const childSkus  = getCol(COL_CHILD_SKU);
    const childLocs  = getCol(COL_CHILD_LOC);
    const qtys       = getCol(COL_QTY);
    const effStarts  = getCol(COL_EFF_START);
    const effEnds    = getCol(COL_EFF_END);
    const bomTypes   = getCol(COL_BOM_TYPE);

    for (let i = 0; i < rowCount; i++) {
      const parentSku = String(parentSkus[i] ?? "").trim();
      const parentLoc = String(parentLocs[i] ?? "").trim();
      const childSku  = String(childSkus[i]  ?? "").trim();
      const childLoc  = String(childLocs[i]  ?? "").trim();
      const qty       = parseFloat(String(qtys[i] ?? "1")) || 1;
      const effStart  = formatDate(effStarts[i] as string | number | null);
      const effEnd    = formatDate(effEnds[i]   as string | number | null);
      const bomType   = String(bomTypes[i] ?? "Primary").trim() || "Primary";

      if (!parentSku || !childSku) continue;

      const key = `${parentSku}|${parentLoc}|${bomType}`;
      if (!adjMap.has(key)) adjMap.set(key, []);
      adjMap.get(key)!.push({ childSku, childLoc, qty, effStart, effEnd });

      allParentSkus.add(parentSku);
      allChildSkus.add(childSku);
    }

    console.log(`  Parsed rows ${startRow}–${startRow + rowCount - 1}`);
  }

  console.log(`Unique parent keys: ${adjMap.size}`);

  // ── 3. Find root nodes ───────────────────────────────────
  // ROOT_PREFIXES: SKUs starting with these are always roots
  // even if they also appear as a child in another BOM.
  // e.g. FG67915-2512 is a child of FG76937-2512 but is still
  // a standalone finished good that needs its own explosion.
  const ROOT_PREFIXES: string[] = ["FG", "FA"];

  const rootSkus = new Set<string>();
  for (const p of allParentSkus) {
    const isPureRoot   = !allChildSkus.has(p);
    const isPrefixRoot = ROOT_PREFIXES.some((pfx: string) => p.startsWith(pfx));
    if (isPureRoot || isPrefixRoot) rootSkus.add(p);
  }

  const rootCombos: { sku: string; loc: string; bomType: string }[] = [];
  for (const [key] of adjMap) {
    const parts   = key.split("|");
    const sku     = parts[0];
    const loc     = parts[1];
    const bomType = parts[2];
    if (rootSkus.has(sku)) {
      rootCombos.push({ sku, loc, bomType });
    }
  }

  console.log(`Root combos to explode: ${rootCombos.length}`);

  // ── 4. Iterative DFS explosion ───────────────────────────
  const flatRows: FlatRow[] = [];

  for (const { sku: rootSku, loc: rootLoc, bomType } of rootCombos) {

    const stack: StackFrame[] = [{
      currentSku:    rootSku,
      currentLoc:    rootLoc,
      cumulativeQty: 1,
      level:         0,
      ancestorPath:  rootSku
    }];

    while (stack.length > 0) {
      const frame = stack.pop()!;

      const edgeKey  = `${frame.currentSku}|${frame.currentLoc}|${bomType}`;
      const children = adjMap.get(edgeKey);
      if (!children || children.length === 0) continue;

      for (const child of children) {

        // Cycle guard
        if (frame.ancestorPath.split("|").includes(child.childSku)) continue;

        const childCumQty = frame.cumulativeQty * child.qty;

        flatRows.push({
          TopLevelSku:        rootSku,
          ChildSku:           child.childSku,
          ImmediateParentSku: frame.currentSku,
          CumulativeQty:      childCumQty,
          Level:              frame.level + 1,
          BOMType:            bomType,
          LocationName:       rootLoc,
          EffStartDate:       child.effStart,
          EffEndDate:         child.effEnd
        });

        const childEdgeKey = `${child.childSku}|${child.childLoc}|${bomType}`;
        if (adjMap.has(childEdgeKey)) {
          stack.push({
            currentSku:    child.childSku,
            currentLoc:    child.childLoc,
            cumulativeQty: childCumQty,
            level:         frame.level + 1,
            ancestorPath:  `${frame.ancestorPath}|${child.childSku}`
          });
        }
      }
    }
  }

  console.log(`Total flat rows generated: ${flatRows.length}`);

  // ── 5. Write to FlatBOM sheet ────────────────────────────
  let outSheet = workbook.getWorksheet("BOM");
  if (!outSheet) {
    outSheet = workbook.addWorksheet("BOM");
  } else {
    const used = outSheet.getUsedRange();
    if (used) used.clear();
    const oldTable = findTableByPrefix(workbook, "FlatBOM");
    if (oldTable) oldTable.delete();
  }

  const HEADERS: string[] = [
    "TopLevelSku",
    "ChildSku",
    "ImmediateParentSku",
    "CumulativeQty",
    "Level",
    "BOMType",
    "LocationName",
    "EffStartDate",
    "EffEndDate"
  ];

  // Write header first
  outSheet.getRangeByIndexes(0, 0, 1, HEADERS.length).setValues([HEADERS]);

  // Write data in chunks of 5000 rows to stay under payload limit
  const WRITE_CHUNK = 5000;
  for (let w = 0; w < flatRows.length; w += WRITE_CHUNK) {
    const slice = flatRows.slice(w, w + WRITE_CHUNK);
    const chunkData: (string | number)[][] = slice.map(row => [
      row.TopLevelSku,
      row.ChildSku,
      row.ImmediateParentSku,
      row.CumulativeQty,
      row.Level,
      row.BOMType,
      row.LocationName,
      row.EffStartDate,
      row.EffEndDate
    ]);
    // +1 offset because row 0 is the header
    outSheet.getRangeByIndexes(w + 1, 0, chunkData.length, HEADERS.length).setValues(chunkData);
    console.log(`  Written rows ${w + 1}–${w + chunkData.length}`);
  }

  // Register full range as named Table for easy downstream querying
  const totalWritten = flatRows.length + 1; // +1 for header
  const tableRange = outSheet.getRangeByIndexes(0, 0, totalWritten, HEADERS.length);
  workbook.addTable(tableRange, true).setName("FlatBOM");

  // ── JnJ Brand Styling ────────────────────────────────────
  // Header row — JnJ red background, white bold text
  const hdrRange = outSheet.getRangeByIndexes(0, 0, 1, 9);
  hdrRange.getFormat().getFill().setColor("1F3864");
  hdrRange.getFormat().getFont().setColor("FFFFFF");
  hdrRange.getFormat().getFont().setBold(true);

  // Alternating row shading — color in chunks of 2 rows (even=blue, odd=white)
  // Apply both colors in bands of 500 rows to stay under payload limits
  const totalFlatRows = flatRows.length;
  const COLOR_CHUNK = 500;
  for (let r = 1; r <= totalFlatRows; r += COLOR_CHUNK * 2) {
    // White rows in this band
    const whiteStart = r;
    const whiteCount = Math.min(COLOR_CHUNK, totalFlatRows - whiteStart + 1);
    if (whiteCount > 0) {
      outSheet.getRangeByIndexes(whiteStart, 0, whiteCount, 9)
        .getFormat().getFill().setColor("FFFFFF");
    }
    // Blue rows in this band
    const blueStart = r + COLOR_CHUNK;
    const blueCount = Math.min(COLOR_CHUNK, totalFlatRows - blueStart + 1);
    if (blueStart <= totalFlatRows && blueCount > 0) {
      outSheet.getRangeByIndexes(blueStart, 0, blueCount, 9)
        .getFormat().getFill().setColor("DAEEF3");
    }
  }

  outSheet.getUsedRange().getFormat().autofitColumns();
  outSheet.activate();

  console.log(`Done! Rows written: ${flatRows.length}`);
  console.log(`Timestamp: ${new Date().toISOString()}`);
}

// ── Helper: normalize date values from Excel ─────────────
function formatDate(val: string | number | null): string {
  if (val === null || val === undefined || val === "") return "";
  if (typeof val === "number") {
    // Excel serial date → convert to ISO string
    const date = new Date(Math.round((val - 25569) * 86400 * 1000));
    return date.toISOString().split("T")[0];
  }
  const s = String(val).trim();
  if (/^\d{4}-\d{2}-\d{2}/.test(s)) return s.substring(0, 10);
  return s;
}

/**
 * Finds the first table whose name starts with the given prefix (case-insensitive).
 * Prevents crashes when tables are renamed (e.g. FlatBOM → FlatBOM6).
 */
function findTableByPrefix(
  workbook: ExcelScript.Workbook,
  prefix: string
): ExcelScript.Table | undefined {
  const lower = prefix.toLowerCase();
  return workbook
    .getTables()
    .find(t => t.getName().toLowerCase().startsWith(lower));
}

/**
 * Finds the first worksheet whose name starts with the given prefix (case-insensitive).
 * Prevents crashes when sheets are renamed (e.g. "BOM Report Output 2" or "BOM Report Output_v2").
 */
function findSheetByPrefix(
  workbook: ExcelScript.Workbook,
  prefix: string
): ExcelScript.Worksheet | undefined {
  const lower = prefix.toLowerCase();
  return workbook
    .getWorksheets()
    .find(s => s.getName().toLowerCase().startsWith(lower));
}
